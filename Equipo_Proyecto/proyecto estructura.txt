-- ============================================
--   CONSULTAS DE VERIFICACIÓN
-- ============================================

-- 1. Todas las compras con detalle de cliente, gerente y producto
SELECT
    c.id            AS id_compra,
    cl.nombre       AS cliente,
    cl.membresia,
    g.nombre        AS gerente,
    p.nombre        AS producto,
    c.cantidad,
    c.total,
    c.metodo_pago,
    c.fecha
FROM Compra c
JOIN Clientes  cl ON c.id_clientes  = cl.id
JOIN Gerente   g  ON c.id_gerente   = g.id
JOIN Productos p  ON c.id_productos = p.id;
GO

-- 3. Inventario de productos con disponibilidad
SELECT
    id,
    nombre,
    precio,
    stock,
    CASE
        WHEN stock > 10 THEN 'Disponible'
        WHEN stock > 0  THEN 'Pocas unidades'
        ELSE                 'Agotado'
    END AS disponibilidad
FROM Productos
ORDER BY stock DESC;
GO

-- 4. Total de ventas por gerente
SELECT
    g.nombre        AS gerente,
    COUNT(c.id)     AS total_compras,
    SUM(c.total)    AS monto_total
FROM Compra c
JOIN Gerente g ON c.id_gerente = g.id
GROUP BY g.nombre
ORDER BY monto_total DESC;
GO

-- 5. Historial de compras por cliente
SELECT
    cl.nombre       AS cliente,
    p.nombre        AS producto,
    c.cantidad,
    c.total,
    c.metodo_pago,
    c.fecha
FROM Compra c
JOIN Clientes  cl ON c.id_clientes  = cl.id
JOIN Productos p  ON c.id_productos = p.id
ORDER BY cl.nombre, c.fecha DESC;
GO
